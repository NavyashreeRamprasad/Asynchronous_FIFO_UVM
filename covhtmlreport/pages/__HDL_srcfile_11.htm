1  `uvm_analysis_imp_decl(_wr)
     2  `uvm_analysis_imp_decl(_rd)
     3  
     4  class scoreboard extends uvm_scoreboard;
     5    `uvm_component_utils(scoreboard)
     6  
     7    uvm_analysis_imp_wr #(wr_tx, scoreboard) wr_imp;
     8    uvm_analysis_imp_rd #(rd_tx, scoreboard) rd_imp;
     9  
     10    bit [`DATA_WIDTH-1:0] fifo_scb[$];
     11    bit [`DATA_WIDTH-1:0] rdata_temp;
     12  
     13    `NEW_COMP
     14  
     15    function void build_phase(uvm_phase phase);
     16      super.build_phase(phase);
     17      wr_imp = new("wr_imp", this);
     18      rd_imp = new("rd_imp", this);
     19    endfunction
     20  
     21    function void write_wr(wr_tx tx);
     22    if (tx.overflow_o) return; 
     23      if (tx.wr_en_i == 1)
     24        fifo_scb.push_front(tx.wdata_i);
     25    endfunction
     26  
     27    function void write_rd(rd_tx tx);
     28    if (tx.underflow_o) return; 
     29      if (tx.rd_en_i == 1) begin
     30        if (fifo_scb.size() == 0)
     31          `uvm_error("SCB", "Read with empty reference FIFO")
     32        else begin
     33          rdata_temp = fifo_scb.pop_back();
     34          common::scb_count++;
     35          if (rdata_temp == tx.rdata_o) common::matching++;
     36          else                          common::mismatch++;
     37        end
     38      end
     39    endfunction
     40  
     41    function void report_phase(uvm_phase phase);
     42      super.report_phase(phase);
     43      if (common::mismatch > 0 )
     44        `uvm_error("SCB", $sformatf("FAILED match=%0d mismatch=%0d", common::matching, common::mismatch))
     45      else
     46        `uvm_info("SCB", $sformatf("PASSED match=%0d mismatch=%0d", common::matching, common::mismatch), UVM_LOW)
     47    endfunction
     48  endclass
