1  class wr_driver extends uvm_driver#(wr_tx);
     2  virtual intf vif;
     3  `uvm_component_utils(wr_driver)
     4  `NEW_COMP
     5  
     6  function void build_phase(uvm_phase phase);
     7  super.build_phase(phase);

     10  
     11  task run_phase(uvm_phase phase);
     12    vif.wr_en_i = 0;
     13    vif.wdata_i = 0;
     14  wait (vif.rst_i === 1'b1);
     15  forever begin
     16  seq_item_port.get_next_item(req);
     17  drive_tx(req);
     18  `uvm_info("WR_DRV",$sformatf("Got item:\n%s",req.sprint()),UVM_LOW);
     19  seq_item_port.item_done();
     20  end
     21  endtask
     22  
     23  task drive_tx(wr_tx tx);
     24      @(posedge vif.wr_clk_i);
     25      vif.wr_en_i = tx.wr_en_i;
     26      if (tx.wr_en_i == 1)
     27        vif.wdata_i = tx.wdata_i;
     28      @(posedge vif.wr_clk_i);
     29      common::wr_drv_count++;
     30      vif.wr_en_i = 0;
     31  endtask
     32  endclass
