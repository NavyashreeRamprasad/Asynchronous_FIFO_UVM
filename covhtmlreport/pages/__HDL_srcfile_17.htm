1  `timescale 1ns/1ps
     2  `define DEPTH 16 
     3  `define DATA_WIDTH  32
     4  `define PTR_WIDTH  $clog2(`DEPTH) 
     5  module async_fifo (
     6      input  wr_clk_i,
     7      input  rd_clk_i,
     8      input  rst_i,
     9      input  wr_en_i,
     10      input  rd_en_i,
     11      input  [`DATA_WIDTH-1:0] wdata_i,
     12      output reg [`DATA_WIDTH-1:0] rdata_o,
     13      output reg full_o,
     14      output reg empty_o,
     15      output reg overflow_o,
     16      output reg underflow_o
     17  );
     18  
     19  integer i;
     20  
     21  // FIFO Memory
     22  reg [`DATA_WIDTH-1:0] fifo [0:`DEPTH-1];
     23  
     24  // Binary pointers
     25  reg [`PTR_WIDTH-1:0] wr_pntr, rd_pntr;
     26  
     27  // Toggle bits
     28  reg wr_toggle, rd_toggle;
     29  
     30  // Pointer synchronization
     31  
     32  // Gray-coded {toggle, pointer} crossing the clock domains
     33  //       (toggle is the MSB so pointer and toggle change together as ONE value)
     34  reg  [`PTR_WIDTH:0] rd_gray_wr_clk;
     35  reg  [`PTR_WIDTH:0] wr_gray_rd_clk;
     36  
     37  // [NEW] Binary values recovered in the receiving domain (flag logic below is unchanged)
     38  wire [`PTR_WIDTH:0] rd_bin_wr_clk = gray2bin(rd_gray_wr_clk);
     39  wire [`PTR_WIDTH:0] wr_bin_rd_clk = gray2bin(wr_gray_rd_clk);
     40  
     41  wire [`PTR_WIDTH-1:0] rd_pntr_wr_clk   = rd_bin_wr_clk[`PTR_WIDTH-1:0];
     42  wire                  rd_toggle_wr_clk = rd_bin_wr_clk[`PTR_WIDTH];
     43  wire [`PTR_WIDTH-1:0] wr_pntr_rd_clk   = wr_bin_rd_clk[`PTR_WIDTH-1:0];
     44  wire                  wr_toggle_rd_clk = wr_bin_rd_clk[`PTR_WIDTH];
     45  
     46  // [NEW] Binary -> Gray: g = b ^ (b >> 1)
     47  function [`PTR_WIDTH:0] bin2gray(input [`PTR_WIDTH:0] b);
     48      bin2gray = b ^ (b >> 1);
     49  endfunction
     50  
     51  // [NEW] Gray -> Binary: each bit is the XOR of all Gray bits at or above it
     52  function [`PTR_WIDTH:0] gray2bin(input [`PTR_WIDTH:0] g);
     53      integer k;
     54      begin
     55          gray2bin[`PTR_WIDTH] = g[`PTR_WIDTH];
     56          for (k = `PTR_WIDTH-1; k >= 0; k = k - 1)
     57              gray2bin[k] = gray2bin[k+1] ^ g[k];
     58      end
     59  endfunction
     60  
     61  // Write Logic
     62  
     63  always @(posedge wr_clk_i or negedge rst_i)
     64  begin
     65      if(rst_i==0) begin
     66          wr_pntr      <= 0;
     67          wr_toggle    <= 0;
     68          overflow_o   <= 0;
     69  
     70          for(i=0;i<`DEPTH;i=i+1)
     71              fifo[i] <= 0;
     72      end
     73      else begin
     74  
     75          overflow_o <= 0;
     76  
     77          if(wr_en_i) begin
     78  
     79              if(full_o)
     80                  overflow_o <= 1;
     81  
     82              else begin
     83  
     84                  fifo[wr_pntr] <= wdata_i;
     85  
     86                  if(wr_pntr == `DEPTH-1) begin
     87                      wr_pntr   <= 0;
     88                      wr_toggle <= ~wr_toggle;
     89                  end
     90                  else
     91                      wr_pntr <= wr_pntr + 1;
     92  
     93              end
     94          end
     95      end
     96  end
     97  
     98  // Read Logic
     99  
     100  always @(posedge rd_clk_i or negedge rst_i)
     101  begin
     102      if(rst_i==0) begin
     103          rd_pntr      <= 0;
     104          rd_toggle    <= 0;
     105          rdata_o      <= 0;
     106          underflow_o  <= 0;
     107      end
     108      else begin
     109  
     110          underflow_o <= 0;
     111  
     112          if(rd_en_i) begin
     113  
     114              if(empty_o)
     115                  underflow_o <= 1;
     116  
     117              else begin
     118  
     119                  rdata_o <= fifo[rd_pntr];
     120  
     121                  if(rd_pntr == `DEPTH-1) begin
     122                      rd_pntr   <= 0;
     123                      rd_toggle <= ~rd_toggle;
     124                  end
     125                  else
     126                      rd_pntr <= rd_pntr + 1;
     127  
     128              end
     129          end
     130      end
     131  end
     132  
     133  // Synchronize Read Pointer into Write Clock
     134  // Two separate regs (pointer, toggle) replaced by one Gray-coded vector
     135  
     136  always @(posedge wr_clk_i or negedge rst_i)
     137  begin
     138      if(rst_i==0) begin
     139          rd_gray_wr_clk <= 0;
     140      end
     141      else begin
     142          rd_gray_wr_clk <= bin2gray({rd_toggle, rd_pntr});
     143      end
     144  end
     145  
     146  // Synchronize Write Pointer into Read Clock
     147  // Two separate regs (pointer, toggle) replaced by one Gray-coded vector
     148  
     149  always @(posedge rd_clk_i or negedge rst_i)
     150  begin
     151      if(rst_i==0) begin
     152          wr_gray_rd_clk <= 0;
     153      end
     154      else begin
     155          wr_gray_rd_clk <= bin2gray({wr_toggle, wr_pntr});
     156      end
     157  end
     158  
     159  // Full Flag
     160  //  now fed by the Gray->binary decoded values above
     161  
     162  always @(*)
     163  begin
     164      if(rst_i==0)
     165          full_o <= 0;
     166      else
     167          full_o <= (wr_pntr == rd_pntr_wr_clk) &&
     168                    (wr_toggle != rd_toggle_wr_clk);
     169  end
     170  
     171  // Empty Flag
     172  // now fed by the Gray->binary decoded values above
     173  
     174  always @(*)
     175  begin
     176      if(rst_i==0)
     177          empty_o <= 1;
     178      else
     179          empty_o <= (wr_pntr_rd_clk == rd_pntr) &&
     180                     (wr_toggle_rd_clk == rd_toggle);
     181  end
     182  
     183  endmodule
