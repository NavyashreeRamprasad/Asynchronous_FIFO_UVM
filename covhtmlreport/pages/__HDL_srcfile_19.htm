1  module fifo_checks (
     2    // clocks and reset
     3    input wr_clk_i,
     4    input rd_clk_i,
     5    input rst_i,
     6    // port signals
     7    input wr_en_i,
     8    input rd_en_i,
     9    input full_o,
     10    input empty_o,
     11    input overflow_o,
     12    input underflow_o,
     13    input wdata_i,
     14    input rdata_o,
     15    // internal signals
     16    input [`PTR_WIDTH-1:0] wr_pntr,
     17    input [`PTR_WIDTH-1:0] rd_pntr,
     18    input wr_toggle,
     19    input rd_toggle,
     20    input [`PTR_WIDTH-1:0] rd_pntr_wr_clk,
     21    input rd_toggle_wr_clk,
     22    input [`PTR_WIDTH-1:0] wr_pntr_rd_clk,
     23    input wr_toggle_rd_clk
     24  );
     25  
     26    // ---------------- FULL (write clock) ----------------
     27    // Pointers match and toggles differ -> full_o must be 1
     28  property p1;
     29    @(posedge wr_clk_i) disable iff (!rst_i)
     30      ((wr_pntr == rd_pntr_wr_clk) && (wr_toggle != rd_toggle_wr_clk)) |-> full_o;
     31  endproperty
     32    a_full: assert property (p1);
     33  
     34    // Write while full -> overflow_o on next cycle
     35  property p2;
     36      @(posedge wr_clk_i) disable iff (!rst_i)
     37      (wr_en_i && full_o) |=> overflow_o;
     38   endproperty 
     39    a_overflow: assert property (p2);
     40  
     41    // ---------------- EMPTY (read clock) ----------------
     42  
     43    // Pointers match and toggles equal -> empty_o must be 1
     44  property p3;
     45       @(posedge rd_clk_i) disable iff (!rst_i)
     46      ((rd_pntr == wr_pntr_rd_clk) && (rd_toggle == wr_toggle_rd_clk)) |-> empty_o;
     47  endproperty
     48  
     49   a_empty: assert property (p3);
     50  
     51    // Read while empty -> underflow_o on next cycle
     52  property p4;
     53      @(posedge rd_clk_i) disable iff (!rst_i)
     54      (rd_en_i && empty_o) |=> underflow_o;
     55   endproperty 
     56    a_underflow: assert property (p4);
     57  
     58  
     59  // if rst ==0 is wr_en or rd_en high
     60  property p5;
     61    @(posedge wr_clk_i) rst_i ==0 |-> !wr_en_i;
     62    endproperty 
     63    wr_en_a: assert property (p5);
     64  
     65  property p8;
     66    @(posedge rd_clk_i) rst_i ==0 |-> !rd_en_i;
     67    endproperty 
     68    rd_en_a: assert property (p8);
     69  
     70  //is wdata and rdata valid
     71  property p6;
     72    @(posedge wr_clk_i) disable iff (!rst_i)
     73    wr_en_i ==1 |-> !($isunknown(wdata_i));
     74   endproperty 
     75    wdata_a: assert property (p6);
     76  property p7;
     77    @(posedge rd_clk_i) disable iff (!rst_i)
     78    rd_en_i ==1 |=> !($isunknown(rdata_o));
     79   endproperty 
     80    rdata_a: assert property (p7);
     81  
     82  endmodule
     83  
     84  
     85  // Attach the checker to the FIFO (no RTL changes needed)
     86  bind async_fifo fifo_checks u_fifo_checks (
     87    .wr_clk_i         (wr_clk_i),
     88    .rd_clk_i         (rd_clk_i),
     89    .rst_i            (rst_i),
     90    .wr_en_i          (wr_en_i),
     91    .rd_en_i          (rd_en_i),
     92    .full_o           (full_o),
     93    .empty_o          (empty_o),
     94    .overflow_o       (overflow_o),
     95    .underflow_o      (underflow_o),
     96    .wdata_i          (wdata_i),
     97    .rdata_o          (rdata_o),
     98    .wr_pntr          (wr_pntr),
     99    .rd_pntr          (rd_pntr),
     100    .wr_toggle        (wr_toggle),
     101    .rd_toggle        (rd_toggle),
     102    .rd_pntr_wr_clk   (rd_pntr_wr_clk),
     103    .rd_toggle_wr_clk (rd_toggle_wr_clk),
     104    .wr_pntr_rd_clk   (wr_pntr_rd_clk),
     105    .wr_toggle_rd_clk (wr_toggle_rd_clk)
     106  );
